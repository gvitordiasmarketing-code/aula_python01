import tkinter as tk


janela_main = tk.Tk() 
janela_main.title("minha casa")

janela_main.configure(background="red")
janela_main.minsize(200,200)
janela_main.maxsize(500,500)
janela_main.geometry("300x300")

#objetos em janela
tk.Label(janela_main, 
         text="hello word",
         bg="red",
         font=("Arial" ,20 "bold")
         ).pack()
tk.Label(janela_main, 
         text="Gabriel Vitor Dias Rocha",
         bg="silver",
         font=("Arial" ,20)
         ).pack()

#imagens
imagem = tk.PhotoImage(file="galo_time.jpg")
tk.Label(janela_main, image=imagem).pack()



janela_main.mainloop()
